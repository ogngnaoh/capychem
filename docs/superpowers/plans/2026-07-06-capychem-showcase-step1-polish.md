"""Generate real RDKit 2D depictions for the demo's molecules, byte-identical to the
live GET /molecule/svg endpoint (same render_svg). Writes a generated TS module the
Playwright fixtures import. N6: these are in-silico structures, no efficacy claim.

Run: uv run python scripts/gen_demo_molecule_svgs.py
"""

from __future__ import annotations

import json
from pathlib import Path

from capychem.api.routes_molecule import render_svg

# The 9 distinct SMILES the demo renders (TOP_K + Compare GA-268 top_k). Keep in sync
# with apps/web/e2e/_fixtures.ts if those fixtures change.
SMILES = [
    "CC(C)C[C@H](NC(=O)[C@@H](N)Cc1ccccc1)C(=O)N[C@@H](Cc1ccc(O)cc1)C(=O)N[C@@H](CCCCN)C(=O)O",
    "O=C(Nc1ccc(F)cc1)c1ccc2[nH]ncc2c1",
    "COc1ccc(CNc2ncnc3[nH]ccc23)cc1",
    "Cc1cc(Nc2ncnc3sccc23)ccc1Cl",
    "O=C(Nc1cccnc1)c1ccc2nc[nH]c2c1",
    "COc1ccc2nc(N)sc2c1C(=O)Nc1ccccc1",
    "O=C(Nc1ccc(Cl)cc1)c1ccc2[nH]ncc2c1",
    "Cc1ccc(Nc2ncnc3sccc23)cc1",
    "COc1ccc(Nc2ncnc3cccc23)cc1",
]
# Aspirin — fallback for any SMILES the demo requests but we didn't pre-generate.
FALLBACK_SMILES = "CC(=O)Oc1ccccc1C(=O)O"
SIZE = 220

OUT = Path(__file__).resolve().parents[1] / "apps/web/e2e/_molecule-svgs.generated.ts"


def main() -> None:
    entries = {s: render_svg(s, SIZE, SIZE) for s in SMILES}
    fallback = render_svg(FALLBACK_SMILES, SIZE, SIZE)
    lines = [
        "// GENERATED by scripts/gen_demo_molecule_svgs.py — do not edit by hand.",
        "// Real RDKit 2D depictions (apps/api render_svg, byte-identical to GET /molecule/svg).",
        "// N6: in-silico structures only, no real-world efficacy claim.",
        "export const MOLECULE_SVG_BY_SMILES: Record<string, string> = {",
    ]
    for smiles, svg in entries.items():
        lines.append(f"  {json.dumps(smiles)}: {json.dumps(svg)},")
    lines.append("};")
    lines.append("")
    lines.append(f"export const MOLECULE_SVG_FALLBACK: string = {json.dumps(fallback)};")
    lines.append("")
    OUT.write_text("\n".join(lines), encoding="utf-8")
    print(f"wrote {OUT} ({len(entries)} structures + fallback)")


if __name__ == "__main__":
    main()
```

// A soft coral cursor + click ripple, injected for VIDEO captures only (Playwright
// records no native cursor). pointer-events:none so it never intercepts a click; on-token
// coral accent. Re-mounts on every navigation via addInitScript.
export async function installCursor(page: Page) {
  await page.addInitScript(() => {
    const CORAL = "#d97757";
    function mount() {
      if (document.getElementById("__demo_cursor")) return;
      const dot = document.createElement("div");
      dot.id = "__demo_cursor";
      Object.assign(dot.style, {
        position: "fixed", left: "0", top: "0", width: "18px", height: "18px",
        marginLeft: "-9px", marginTop: "-9px", borderRadius: "50%",
        border: `2px solid ${CORAL}`, background: "rgba(217,119,87,0.18)",
        zIndex: "2147483647", pointerEvents: "none", opacity: "0",
        transition: "opacity 120ms ease, transform 80ms ease",
      });
      document.body.appendChild(dot);
      window.addEventListener("mousemove", (e) => {
        dot.style.opacity = "1";
        dot.style.transform = `translate(${e.clientX}px, ${e.clientY}px)`;
      }, { passive: true });
      window.addEventListener("mousedown", (e) => {
        const r = document.createElement("div");
        Object.assign(r.style, {
          position: "fixed", left: `${e.clientX}px`, top: `${e.clientY}px`,
          width: "10px", height: "10px", marginLeft: "-5px", marginTop: "-5px",
          borderRadius: "50%", border: `2px solid ${CORAL}`,
          zIndex: "2147483646", pointerEvents: "none", opacity: "0.9",
          transform: "scale(1)", transition: "transform 500ms ease-out, opacity 500ms ease-out",
        });
        document.body.appendChild(r);
        requestAnimationFrame(() => { r.style.transform = "scale(6)"; r.style.opacity = "0"; });
        setTimeout(() => r.remove(), 520);
      }, { passive: true });
    }
    if (document.body) mount();
    else window.addEventListener("DOMContentLoaded", mount);
  });
}
```

- [ ] **Step 1: Write the conversion script**

`scripts/convert-demo-clips.sh`:

```bash
#!/usr/bin/env bash
# Convert the 6 demo clip videos (apps/web/test-results/**/video.webm) into GIFs under
# docs/assets/ via the two-pass palette (fps=10, 880px). Run AFTER:
#   cd apps/web && pnpm exec playwright test e2e/demo.spec.ts
set -euo pipefail
ROOT="$(cd "$(dirname "$0")/.." && pwd)"
RESULTS="$ROOT/apps/web/test-results"
ASSETS="$ROOT/docs/assets"
TMP="$(mktemp -d)"

MAP=(
  "demo-hero:demo"
  "demo-clip-1-launch-resolve:clip-launch-resolve"
  "demo-clip-2-live-cockpit:clip-live-cockpit"
  "demo-clip-3-beat-graph-ga:clip-beat-graph-ga"
  "demo-clip-4-make-it-yours:clip-make-it-yours"
  "demo-clip-5-quality-export:clip-quality-export"
)

for pair in "${MAP[@]}"; do
  slug="${pair%%:*}"; out="${pair##*:}"
  webm="$(find "$RESULTS" -path "*${slug}*" -name '*.webm' | head -1)"
  if [ -z "$webm" ]; then echo "MISSING webm for $slug" >&2; exit 1; fi
  ffmpeg -y -i "$webm" -vf "fps=10,scale=880:-1:flags=lanczos,palettegen=stats_mode=diff" "$TMP/$out.png" -loglevel error
  ffmpeg -y -i "$webm" -i "$TMP/$out.png" -filter_complex \
    "fps=10,scale=880:-1:flags=lanczos[x];[x][1:v]paletteuse=dither=bayer:bayer_scale=5" \
    "$ASSETS/$out.gif" -loglevel error
  echo "wrote $ASSETS/$out.gif ($(du -h "$ASSETS/$out.gif" | cut -f1))"
done
rm -rf "$TMP"
```
